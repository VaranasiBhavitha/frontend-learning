CREATE DATABASE bookstore;
USE bookstore;

CREATE TABLE book
(
    id INT PRIMARY KEY AUTO_INCREMENT,
    category VARCHAR(50),
    book_name VARCHAR(100),
    author VARCHAR(100),
    publication VARCHAR(100),
    price DOUBLE,
    quantity INT,
    amount DOUBLE
);
